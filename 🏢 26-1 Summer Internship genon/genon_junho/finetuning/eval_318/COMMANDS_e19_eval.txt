# 1-1. dp500: 매니페스트 image_path 를 절대경로로 (evaluate_unified 가 test_data 폴더기준
#       상대경로로 이미지를 찾기 때문에, 원본 위치와 다른 곳에서 돌리려면 절대경로화 필수)
python3 - <<'PY'
import json
src="/home/jhyeo/tsr_eval/data/500_genos_val_set/inference_manifest_noocr.jsonl"
root="/home/jhyeo/tsr_eval/data/500_genos_val_set/"
out="/home/jhyeo/finetuning/eval_318/_infer_shards/dp500_full_abs.jsonl"
with open(src) as f, open(out,"w") as w:
    for line in f:
        r=json.loads(line); r["image_path"]=root+r["image_path"]
        w.write(json.dumps(r,ensure_ascii=False)+"\n")
PY

# 1-2. tsr200: output_format=html 강제 + image_path 절대경로화
#       (test.jsonl 은 output_format 이 None 이라 그냥 넣으면 JSON 모드로 잘못 채점됨)
python3 - <<'PY'
import json, os
src="/home/jhyeo/tsr_eval/data/20260317_4_6000/data_split/test.jsonl"
base="/home/jhyeo/tsr_eval/data/20260317_4_6000/data_split"
out="/home/jhyeo/finetuning/eval_318/_infer_shards/tsr200_html.jsonl"
with open(src) as f, open(out,"w") as w:
    for line in f:
        r=json.loads(line); r["output_format"]="html"
        r["image_path"]=os.path.normpath(os.path.join(base, r["image_path"]))
        w.write(json.dumps(r,ensure_ascii=False)+"\n")
PY

[2] (LoRA 어댑터 → 통짜 모델 merge)  ※ 왜 필요한가
--------------------------------------------------------------------------------
# vLLM 0.18.1 이 우리 LoRA 어댑터를 런타임 로딩할 때 set_lora IndexError(fused qkv/gate_up)
# 로 죽어서, 어댑터를 베이스에 미리 합쳐(merge) 통짜 모델로 추론한다.
# (merged 모델은 임시라 추론 끝나면 삭제)
CUDA_VISIBLE_DEVICES=0 /home/jhyeo/finetuning/vlm/.venv/bin/python - <<'PY'
import os, torch
from peft import PeftModel
from transformers import AutoModelForImageTextToText, AutoProcessor
base="/home/jhyeo/ocr_file_filter/_models/Qwen3.5-9B"
adapter="/home/jhyeo/finetuning/vlm/output/e19_qwen35_9b_37818/student_sft/checkpoint-<STEP>"
out="/home/jhyeo/finetuning/vlm/output/e19_qwen35_9b_37818/_sweep_merged/ckpt-<STEP>"
m=AutoModelForImageTextToText.from_pretrained(base,torch_dtype=torch.bfloat16,trust_remote_code=True,device_map="auto")
m=PeftModel.from_pretrained(m,adapter).merge_and_unload()
os.makedirs(out,exist_ok=True); m.save_pretrained(out)
AutoProcessor.from_pretrained(base,trust_remote_code=True).save_pretrained(out)
print("MERGED:",out)
PY
