# 1-1. dp500 (NO-OCR): image_path 절대경로화 + bbox_space=image_px + 실제 W/H 주입
/home/jhyeo/finetuning/vlm/.venv/bin/python - <<'PY'
import json
from PIL import Image
src="/home/jhyeo/tsr_eval/data/500_genos_val_set/inference_manifest_noocr.jsonl"
root="/home/jhyeo/tsr_eval/data/500_genos_val_set/"
out="/home/jhyeo/finetuning/eval_318/_infer_shards_e20/dp500_full_abs.jsonl"
with open(src) as f, open(out,"w") as w:
    for line in f:
        r=json.loads(line); ip=root+r["image_path"]; r["image_path"]=ip
        wid,hei=Image.open(ip).size
        r["bbox_space"]="image_px"; r["image_width"]=wid; r["image_height"]=hei
        r["bbox_scale"]=None
        w.write(json.dumps(r,ensure_ascii=False)+"\n")
print("dp500 ->",out)
PY

# 1-2. tsr200 (WITH-OCR): output_format=html + 절대경로 + bbox_space=image_px + W/H
/home/jhyeo/finetuning/vlm/.venv/bin/python - <<'PY'
import json, os
from PIL import Image
src="/home/jhyeo/tsr_eval/data/20260317_4_6000/data_split/test.jsonl"
base="/home/jhyeo/tsr_eval/data/20260317_4_6000/data_split"
out="/home/jhyeo/finetuning/eval_318/_infer_shards_e20/tsr200_html.jsonl"
with open(src) as f, open(out,"w") as w:
    for line in f:
        r=json.loads(line); r["output_format"]="html"
        ip=os.path.normpath(os.path.join(base, r["image_path"])); r["image_path"]=ip
        wid,hei=Image.open(ip).size
        r["bbox_space"]="image_px"; r["image_width"]=wid; r["image_height"]=hei
        r["bbox_scale"]=None
        w.write(json.dumps(r,ensure_ascii=False)+"\n")
print("tsr200 ->",out)
PY

# 예: GPU0=200,3700 / GPU2=500,3800 / GPU3=1000,4000
# 아래를 3개 tmux 창(또는 &)으로 GPU 만 바꿔 실행. merge→dp500→tsr200 순서.
run_ckpts() {  # $1=gpu  $2..=steps
  local gpu=$1; shift
  cd /home/jhyeo/finetuning/vlm
  for STEP in "$@"; do
    CUDA_VISIBLE_DEVICES=$gpu /home/jhyeo/finetuning/vlm/.venv/bin/python - "$STEP" <<'PY'
import os,sys,torch
from peft import PeftModel
from transformers import AutoModelForImageTextToText, AutoProcessor
STEP=sys.argv[1]; base="/home/jhyeo/ocr_file_filter/_models/Qwen3.5-9B"
ad=f"/home/jhyeo/finetuning/vlm/output/e20_qwen35_9b_pixel_bboxw/student_sft/checkpoint-{STEP}"
out=f"/home/jhyeo/finetuning/vlm/output/e20_qwen35_9b_pixel_bboxw/_sweep_merged/ckpt-{STEP}"
m=AutoModelForImageTextToText.from_pretrained(base,torch_dtype=torch.bfloat16,trust_remote_code=True,device_map="auto")
PeftModel.from_pretrained(m,ad).merge_and_unload().save_pretrained(out)
AutoProcessor.from_pretrained(base,trust_remote_code=True).save_pretrained(out); print("MERGED",out)
PY
    MERGED=output/e20_qwen35_9b_pixel_bboxw/_sweep_merged/ckpt-$STEP
    INFER=/home/workspace/tsr_test/.venv/bin/python
    SH=/home/jhyeo/finetuning/eval_318/_infer_shards_e20
    CUDA_VISIBLE_DEVICES=$gpu $INFER -m eval.evaluate_unified --backend vllm \
      --test_data $SH/dp500_full_abs.jsonl \
      --output_dir eval_results/e20_sweep_dp500_ckpt$STEP \
      --model $MERGED --max_model_len 16384 --batch_size 16 --no-thinking
    CUDA_VISIBLE_DEVICES=$gpu $INFER -m eval.evaluate_unified --backend vllm \
      --test_data $SH/tsr200_html.jsonl \
      --output_dir eval_results/e20_sweep_tsr200_ckpt$STEP \
      --model $MERGED --max_model_len 16384 --batch_size 16 --no-thinking
  done
}
# tmux 3창:  run_ckpts 0 200 3700   /   run_ckpts 2 500 3800   /   run_ckpts 3 1000 4000
